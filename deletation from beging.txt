#include <stdio.h>

#define MAX 10

int data[MAX];
int next[MAX];
int head = -1;
int freeIndex = 0;

void insertEnd(int value)
{
    int newIndex = freeIndex;

    data[newIndex] = value;
    next[newIndex] = -1;

    if (head == -1)
    {
        head = newIndex;
    }
    else
    {
        int temp = head;

        while (next[temp] != -1)
        {
            temp = next[temp];
        }

        next[temp] = newIndex;
    }

    freeIndex++;
}

void deleteBeginning()
{
    int temp;

    if (head == -1)
    {
        printf("List is empty!\n");
        return;
    }

    temp = head;
    head = next[head];

    printf("Deleted element: %d\n", data[temp]);
}

void display()
{
    int temp = head;

    while (temp != -1)
    {
        printf("%d -> ", data[temp]);
        temp = next[temp];
    }

    printf("NULL\n");
}

int main()
{
    insertEnd(10);
    insertEnd(20);
    insertEnd(30);

    printf("Before deletion:\n");
    display();

    deleteBeginning();

    printf("After deletion:\n");
    display();

    return 0;
}
